    user_id BIGINT,
    event_type VARCHAR(32),
    product_id INTEGER,
    zaman TIMESTAMPTZ
);

DROP TABLE IF EXISTS items;

CREATE TABLE items (
    product_id INTEGER,
    category_id BIGINT,
    category_code VARCHAR(255),
    brand TEXT
);



create table groups as
SELECT *, LAG(zaman)
over (partition by user_id order by zaman)
as prev_zaman from events;



CREATE TABLE clean AS
SELECT *
FROM groups
WHERE prev_zaman IS NULL
   OR zaman - prev_zaman > INTERVAL '1 second';
